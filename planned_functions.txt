
# Locations exceeding a criteria set in the current monitoring round.
#
# `type` picks which column of `analytes` supplies the threshold ("eco" or "hh").
# Analytes with no criteria of that type (NA in the tibble) are skipped.
#
# Returns:
#   $features - the exceeding rows themselves, one row per exceeding sample, with the
#               analyte `key` and the `threshold` applied joined on
#   $locs     - named list of sorted, de-duplicated location_code vectors, keyed by
#               analyte key: $locs$PFOS, $locs$PFOA, $locs$SumPFAS
#
# Replaces get_eco_exceedances()/get_hh_exceedances(), which were the same function with
# different criteria and built their location vectors via a pivot_longer -> pivot_wider
# (values_from = location_code) trick that produced list-columns and a "values are not
# uniquely identified" warning.
get_exceedances <- function(data, type = c("eco", "hh"), round = round_id) {
  type <- match.arg(type)

  criteria <- analytes %>%
    select(key, chem_code, threshold = all_of(type)) %>%
    filter(!is.na(threshold))

  features <- data %>%
    filter(chem_code %in% criteria$chem_code, monitoring_round == round) %>%
    inner_join(criteria, by = "chem_code") %>%
    # is.na(prefix) marks a detection - a "<LOR" result is never an exceedance
    filter(is.na(prefix), concentration > threshold) %>%
    mutate(criteria_type = type)

  list(
    features = features,
    locs = set_names(criteria$key) %>%
      map(\(k) {
        features %>%
          filter(key == k) %>%
          pull(location_code) %>%
          unique() %>%
          sort()
      })
  )
}

# Per-group historical min/max and the current-round result, as a one-row tibble.
#
# Conventions (unchanged from the original, but now applied symmetrically):
#   * the historical MINIMUM is taken over all prior results, detections or not;
#   * the historical MAXIMUM is taken over prior DETECTIONS only, because the LOR has
#     fallen over the life of the program and an old high "<LOR" would otherwise dominate;
#   * the current result is the MAXIMUM of the round, not whichever row sorted first.
#
# Returns NA rather than the Inf/-Inf that min()/max() give on an empty vector - a
# location sampled for the first time this round has no history at all.
range_stats <- function(concentration, prefix, monitoring_round, round) {
  hist        <- monitoring_round != round
  hist_detect <- hist & prefix == ""
  cur         <- monitoring_round == round

  # index back into the FULL vectors so the prefix always belongs to the value reported
  min_i <- if (any(hist))        which(hist)[which.min(concentration[hist])]               else NA_integer_
  max_i <- if (any(hist_detect)) which(hist_detect)[which.max(concentration[hist_detect])] else NA_integer_
  cur_i <- if (any(cur))         which(cur)[which.max(concentration[cur])]                 else NA_integer_

  at <- function(v, i, empty) if (is.na(i)) empty else v[i]

  tibble(
    # every result held for the group, this round included - the number of samples the
    # range beside it is drawn from
    n_samples             = length(concentration),
    min_conc              = at(concentration, min_i, NA_real_),
    max_conc              = at(concentration, max_i, NA_real_),
    min_prefix            = at(prefix,        min_i, NA_character_),
    max_prefix            = at(prefix,        max_i, NA_character_),
    current_concentration = at(concentration, cur_i, NA_real_),
    current_prefix        = at(prefix,        cur_i, NA_character_),
    n_current             = sum(cur)
  )
}

# Adds the new-maximum / new-minimum / order-of-magnitude flags to a range_stats() result.
#
# A new MAXIMUM is only meaningful between two DETECTIONS. max_conc already carries that
# restriction on the historical side; the current side is imposed here, so a "<0.05" result
# can never be reported as a new maximum however high the LOR of the day happened to be.
# A new MINIMUM keeps the range_stats convention of using all prior results, detections or
# not, so a non-detect can legitimately set one.
#
# max_ratio is the current result as a multiple of the previous detected maximum, and
# `spike` marks the ones above `factor` (default 10x) - a jump that large is more often a
# transcription or unit problem than a real change, so it is worth checking against the lab
# report before any text is written around it.
flag_new_max <- function(df, factor = 10) {
  df %>%
    mutate(
      # replace_na: a location not sampled this round has NA here, and gt cannot use an
      # NA as a row-selection predicate.
      new_max   = replace_na(current_prefix == "" & current_concentration > max_conc, FALSE),
      new_min   = replace_na(current_concentration < min_conc, FALSE),
      max_ratio = if_else(new_max & max_conc > 0, current_concentration / max_conc, NA_real_),
      spike     = replace_na(new_max & max_ratio > factor, FALSE)
    )
}

# show_n adds a sample-count column beside the group: the total number of results held for
# that group and analyte, current round included.
#
# spike_factor shades the current concentration where it is a new maximum more than that
# many times the previous detected maximum (NULL switches the check off); spike_only cuts
# the table down to those rows.
#
# exceedance_threshold cuts the table to this round's exceedances. An exceedance is a
# DETECTED result above the threshold - current_prefix == "" - matching get_exceedances(),
# exceedance_map_pair() and the interactive maps, so no table can report an exceedance a
# map does not draw. Set exceedance_detects_only = FALSE where the threshold is being used
# as a plain "has a current result above x" cut rather than as a criteria; the surface
# water call below passes 0 for exactly that reason and needs its <LOR rows kept.
historical_range_table <- function(data, key, group_vars = "location_code",
                                   exceedance_threshold = NULL, highlight_new = TRUE,
                                   show_n = FALSE, round = round_id,
                                   spike_factor = 10, spike_only = FALSE,
                                   exceedance_detects_only = TRUE) {
  df <- data %>%
    filter(chem_code == pfas_codes[[key]]) %>%
    replace_na(list(prefix = "")) %>%
    group_by(across(all_of(group_vars))) %>%
    reframe(range_stats(concentration, prefix, monitoring_round, round))

  # The raw feed is known to contain duplicates; say so rather than silently picking one.
  if (any(df$n_current > 1)) {
    warning(
      "historical_range_table(", key, "): more than one result in ", round,
      " for: ", paste(df$location_code[df$n_current > 1], collapse = ", "),
      ". Reporting the maximum.",
      call. = FALSE
    )
  }
  df <- df %>% select(-n_current)
  if (!show_n) df <- df %>% select(-n_samples)

  if (!is.null(exceedance_threshold)) {
    df <- df %>% filter(current_concentration > exceedance_threshold)
    # a "<LOR" result is never an exceedance, however high the reported limit
    if (exceedance_detects_only) df <- df %>% filter(current_prefix == "")
  }

  # Computed unconditionally: the same flags drive both the bolding and the spike check.
  df <- flag_new_max(df, factor = spike_factor %||% Inf)
  if (spike_only) df <- df %>% filter(spike)

  tbl <- df %>%
    table_style() %>%
    sub_missing(columns = current_concentration, missing_text = "Not Sampled") %>%
    sub_missing(columns = current_prefix, missing_text = "") %>%
    # a location first sampled this round has no history at all
    sub_missing(columns = c(min_conc, max_conc), missing_text = "No History") %>%
    sub_missing(columns = c(min_prefix, max_prefix), missing_text = "") %>%
    fmt_number(columns = c(min_conc, max_conc, current_concentration), decimals = 5, drop_trailing_zeros = TRUE) %>%
    cols_merge(columns = c(min_prefix, min_conc), pattern = "{1}{2}") %>%
    cols_merge(columns = c(max_prefix, max_conc), pattern = "{1}{2}") %>%
    cols_merge(columns = c(current_prefix, current_concentration), pattern = "{1}{2}") %>%
    cols_merge_range(col_begin = min_prefix, col_end = max_prefix, sep = " - ") %>%
    cols_label(location_code = "Location", min_prefix = "Historical Range", current_prefix = "Current Concentration") %>%
    cols_align(align = "right")

  if (show_n) {
    tbl <- tbl %>%
      fmt_integer(columns = n_samples) %>%
      cols_label(n_samples = "Samples (n)") |> 
      gt::cols_move_to_end(n_samples)
  }

  if (highlight_new) {
    tbl <- tbl %>%
      tab_style(style = cell_text(weight = "bold"),
                locations = cells_body(columns = current_prefix, rows = new_max == TRUE)) %>%
      tab_style(style = cell_text(style = "italic", decorate = "underline"),
                locations = cells_body(columns = current_prefix, rows = new_min == TRUE))
  }

  if (!is.null(spike_factor) && any(df$spike)) {
    tbl <- tbl %>%
      tab_style(style = cell_fill(color = "#f8d7da"),
                locations = cells_body(columns = current_prefix, rows = spike == TRUE)) %>%
      tab_footnote(
        footnote  = paste0("Shaded: new maximum more than ", spike_factor,
                           "x the previous detected maximum - verify against the lab report."),
        locations = cells_column_labels(columns = current_prefix)
      )
  }

  tbl %>% cols_hide(c(new_min, new_max, max_ratio, spike))
}

# Standalone screening version of the same check, run across every analyte at once rather
# than one key at a time, for the "New Maximums" sections.
#
# spikes_only = FALSE widens it to every new maximum, spike or not.
new_max_check <- function(data, factor = 10, group_vars = c("location_code", "chem_name"),
                          round = round_id, spikes_only = TRUE, decimals = 5) {
  df <- data %>%
    filter(chem_code %in% analytes$chem_code) %>%
    replace_na(list(prefix = "")) %>%
    group_by(across(all_of(group_vars))) %>%
    reframe(range_stats(concentration, prefix, monitoring_round, round)) %>%
    flag_new_max(factor = factor) %>%
    filter(if (spikes_only) spike else new_max) %>%
    arrange(desc(max_ratio)) %>%
    select(all_of(group_vars), n_samples, max_prefix, max_conc,
           current_prefix, current_concentration, max_ratio)

  if (nrow(df) == 0) {
    message("new_max_check(): no detection in ", round, " exceeds ", factor,
            "x the previous detected maximum.")
  }

  labels <- c(location_code  = "Location",
              monitoring_zone = "Zone",
              chem_name      = "Analyte",
              n_samples      = "Samples (n)",
              max_prefix     = "Previous Maximum (detects)",
              current_prefix = "Current Concentration",
              max_ratio      = "x Previous Max")

  df %>%
    table_style() %>%
    fmt_number(columns = c(max_conc, current_concentration), decimals = decimals,
               drop_trailing_zeros = TRUE) %>%
    fmt_number(columns = max_ratio, decimals = 1) %>%
    cols_merge(columns = c(max_prefix, max_conc), pattern = "{1}{2}") %>%
    cols_merge(columns = c(current_prefix, current_concentration), pattern = "{1}{2}") %>%
    cols_label(!!!as.list(labels[names(labels) %in% names(df)])) %>%
    cols_align(align = "right")
}

boxplot_with_criteria <- function(plot, criteria, filename, width = 18, height = 14, dpi = 300) {
  p <- plot +
    # geom_vline's default key glyph is a vertical stroke; "path" draws it horizontally.
    geom_vline(aes(xintercept = criteria, color = "Criteria"), linetype = "dashed", lwd = 1,
               key_glyph = "path") +
    scale_color_manual(name = "Legend Title", values = c("Criteria" = "red")) +
    # Match the key glyph to the line it stands for: dashed red, no point marker.
    guides(color = guide_legend(override.aes = list(linetype = "dashed", shape = NA)))
  ggsave(filename, plot = p, width = width, height = height, units = "cm", dpi = dpi)
  p
}

# ---------------------------------------------------------------------------------------
# Summary table of samples / detects / concentration range, one row per analyte.
#
# Replaces four near-identical ~50-line blocks (groundwater, surface water, sediment,
# pore water) that had drifted apart in how they computed the maximum and how they
# attached exceedance locations.
# ---------------------------------------------------------------------------------------

# Report the maximum DETECTED concentration. Where nothing was detected there is no
# detection to report, so fall back to the highest result so the "<" is carried through
# honestly rather than the row coming out blank.
summary_stats <- function(concentration, prefix, location_code) {
  detect   <- is.na(prefix)
  max_pool <- if (any(detect)) which(detect) else seq_along(concentration)

  min_i <- which.min(concentration)
  max_i <- max_pool[which.max(concentration[max_pool])]
  at <- function(v, i, empty) if (!length(i)) empty else v[i]

  tibble(
    n_samples  = length(concentration),
    detects    = sum(detect),
    min_conc   = at(concentration,  min_i, NA_real_),
    max_conc   = at(concentration,  max_i, NA_real_),
    min_prefix = at(prefix,         min_i, NA_character_),
    max_prefix = at(prefix,         max_i, NA_character_),
    max_loc    = at(location_code,  max_i, NA_character_)
  )
}

# exceedances: named list of the $locs elements returned by get_exceedances(), e.g.
#   list(HH = hh$locs, ECO = eco$locs)
# Each becomes a "<name>_Exceedance" column, matched to analytes by key rather than by
# row position - the old cols_add() calls assumed exactly three rows in alphabetical order.
#
# show_exceedance_n adds a "<name>_Exceedance_n" count column beside each of those, giving
# the number of DISTINCT LOCATIONS exceeding that criteria set for that analyte (not the
# number of exceeding results - $locs is already de-duplicated by get_exceedances()).
analyte_summary_table <- function(data,
                                  exceedances       = list(),
                                  show_max_loc      = FALSE,
                                  show_exceedance_n = FALSE,
                                  round             = round_id,
                                  table_width       = 1000,
                                  decimals          = 5) {
  df <- data %>%
    filter(chem_code %in% analytes$chem_code, monitoring_round == round) %>%
    left_join(analytes %>% select(key, chem_code), by = "chem_code") %>%
    group_by(key, chem_name, output_unit) %>%
    reframe(summary_stats(concentration, prefix, location_code)) %>%
    arrange(match(key, analytes$key))

  for (nm in names(exceedances)) {
    locs <- exceedances[[nm]]
    df[[paste0(nm, "_Exceedance")]] <-
      map_chr(df$key, \(k) paste0(locs[[k]] %||% character(0), collapse = ", "))

    # NA, not 0, where the analyte has no criteria of this type at all (PFOS has no HH
    # value, SumPFAS no ECO value) - sub_missing() blanks it, matching the empty location
    # cell beside it. A real 0 means "criteria apply, nothing exceeded them".
    if (show_exceedance_n) {
      df[[paste0(nm, "_Exceedance_n")]] <-
        map_int(df$key, \(k) if (is.null(locs[[k]])) NA_integer_ else length(locs[[k]]))
    }
  }

  tbl <- df %>%
    select(-key) %>%
    table_style() %>%
    sub_missing(missing_text = "") %>%
    fmt_number(decimals = decimals, drop_trailing_zeros = TRUE) %>%
    cols_merge(columns = c(min_prefix, min_conc), pattern = "{1}{2}") %>%
    cols_merge(columns = c(max_prefix, max_conc), pattern = "{1}{2}") %>%
    cols_label(
      chem_name  = "Analyte",
      min_prefix = "Minimum Concentration",
      max_prefix = "Maximum Concentration",
      max_loc    = "Maximum Concentration Location"
    ) %>%
    cols_align(align = "right")

  if (!show_max_loc)         tbl <- tbl %>% cols_hide(max_loc)
  if (!is.null(table_width)) tbl <- tbl %>% tab_options(table.width = table_width)
  tbl
}

# The "highest / lowest concentration per analyte" pair, repeated for SW, SED and PW.
min_max_locations_table <- function(data, n_max = 1, n_min = 4,
                                    round = round_id, decimals = 5) {
  base <- data %>%
    filter(monitoring_round == round, chem_code %in% analytes$chem_code) %>%
    group_by(chem_name)

  pick <- function(x) {
    x %>%
      select(monitoring_zone, field_id, prefix, concentration) %>%
      arrange(field_id) %>%
      table_style(decimals = decimals)
  }

  list(
    max = pick(slice_max(base, concentration, n = n_max)),
    min = pick(slice_min(base, concentration, n = n_min))
  )
}
